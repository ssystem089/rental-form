function askAvailability() { const model = document.getElementById("model").value; const lens = document.getElementById("lens").value; const pickupValue = document.getElementById("pickupTime").value; const returnDate = document.getElementById("returnDate").value; if (!pickupValue || !returnDate) { alert("請先選擇預計取機時間與歸還日期。"); return; } let equipment = model; if ( lens !== "無加選配件 (單機身)" && lens !== "無適配增距鏡" ) { equipment += ` + ${lens}`; } const pickupTime = pickupValue.replace("T", " "); const inquiryMessage = `您好，想詢問設備是否有檔期： 租借器材：${equipment} 預計取機：${pickupTime} 預計歸還：${returnDate} 請協助確認是否可租，謝謝。`; const officialAccountId = "@你的官方帳號LINE_ID"; const lineUrl = "https://line.me/R/oaMessage/" + encodeURIComponent(officialAccountId) + "/?" + encodeURIComponent(inquiryMessage); window.location.href = lineUrl; }
